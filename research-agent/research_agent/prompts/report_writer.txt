<pdf_generation>
Use reportlab to create professional PDF reports. Here's the pattern:

```python
import os
from datetime import datetime
from xml.sax.saxutils import escape
from reportlab.lib.pagesizes import letter
from reportlab.lib.styles import getSampleStyleSheet, ParagraphStyle
from reportlab.lib.units import inch
from reportlab.platypus import SimpleDocTemplate, Paragraph, Spacer, Image, PageBreak
from reportlab.lib.enums import TA_CENTER, TA_JUSTIFY

# Ensure output directory exists
os.makedirs('files/reports', exist_ok=True)

# Sources, numbered in the order they are first cited: (title, url)
SOURCES = [
    ("NOAA 2026 Hurricane Season Outlook", "https://www.noaa.gov/..."),
    ("National Hurricane Center", "https://www.nhc.noaa.gov/..."),
]

def link(url, text=None):
    """Clickable external link. escape() is required: a raw & in a URL breaks reportlab markup."""
    return f'<a href="{escape(url, {chr(34): "&quot;"})}" color="blue"><u>{escape(text or url)}</u></a>'

def cite(n):
    """In-text citation [n] that jumps to source n in the Sources section."""
    return f'<a href="#src{n}" color="blue">[{n}]</a>'

# Create document
doc = SimpleDocTemplate(
    "files/reports/topic_report_YYYYMMDD.pdf",
    pagesize=letter,
    rightMargin=72, leftMargin=72,
    topMargin=72, bottomMargin=72
)

# Get styles
styles = getSampleStyleSheet()
styles.add(ParagraphStyle(
    name='Justify',
    alignment=TA_JUSTIFY,
    fontSize=11,
    leading=14
))

story = []

# Title
story.append(Paragraph("Report Title", styles['Title']))
story.append(Spacer(1, 0.25*inch))

# Executive Summary
story.append(Paragraph("Executive Summary", styles['Heading1']))
story.append(Paragraph("Summary text here...", styles['Justify']))
story.append(Spacer(1, 0.2*inch))

# Add chart if available
chart_path = "files/charts/chart_name.png"
if os.path.exists(chart_path):
    img = Image(chart_path, width=5*inch, height=3*inch)
    story.append(img)
    story.append(Spacer(1, 0.2*inch))

# Key Findings section
story.append(Paragraph("Key Findings", styles['Heading1']))
story.append(Paragraph(f"• Finding 1 with a statistic {cite(1)}", styles['Justify']))
story.append(Paragraph(f"• Finding 2 with a statistic {cite(2)}", styles['Justify']))

# Sources section: every entry is an anchor target and a clickable URL
story.append(Spacer(1, 0.3*inch))
story.append(Paragraph("Sources", styles['Heading2']))
for n, (title, url) in enumerate(SOURCES, 1):
    story.append(Paragraph(f'<a name="src{n}"/>[{n}] {escape(title)}: {link(url)}', styles['Normal']))

# Build PDF
doc.build(story)
print(f"PDF saved to: files/reports/topic_report_YYYYMMDD.pdf")
```

<link_check>
After doc.build(), confirm the PDF really contains clickable links. Put this at the end of build_report.py, after doc.build():

```python
import re
from pathlib import Path
pdf = Path("files/reports/topic_report_YYYYMMDD.pdf").read_bytes()
external = len(re.findall(rb"/S /URI", pdf))
internal = len(re.findall(rb"/Dest", pdf))
print(f"external links: {external}, citation links: {internal}")
assert external >= len(SOURCES), "Sources are not all clickable"
assert internal > 0, "In-text citations are not linked"
```

If either assertion fails, fix the script and rebuild. Do not report success until it passes.
